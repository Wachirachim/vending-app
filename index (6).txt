function loadImageFile(file){
  return new Promise((resolve, reject) => {
    const url = URL.createObjectURL(file);
    const img = new Image();
    img.onload = () => {
      const maxDim = 2000;
      if(img.width > maxDim || img.height > maxDim){
        let w = img.width, h = img.height;
        if(w > h){ h = Math.round((h * maxDim) / w); w = maxDim; }
        else { w = Math.round((w * maxDim) / h); h = maxDim; }
        const oc = document.createElement('canvas');
        oc.width = w; oc.height = h;
        const octx = oc.getContext('2d');
        octx.drawImage(img, 0, 0, w, h); URL.revokeObjectURL(url);
        const resized = new Image();
        resized.onload = () => resolve(resized);
        resized.onerror = () => resolve(img);
        resized.src = oc.toDataURL('image/jpeg', 0.90);
      } else {
        resolve(img);
      }
    };
    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('โหลดรูปไม่ได้: ' + file.name)); };
    img.src = url;
  });
}

// วาดตราประทับเวลา (Timestamp Watermark) มุมขวาล่าง
function drawTimestampWatermark(ctx, cW, cH, chkId, center = false){
  if(!$(chkId)?.checked) return;
  const now = new Date();
  const dStr = `${pad2(now.getDate())}/${pad2(now.getMonth()+1)}/${now.getFullYear()+543}  ${pad2(now.getHours())}:${pad2(now.getMinutes())} น.`;
  ctx.save();
  const scale = Math.min(cW, cH) / 1200;
  const fontSize = Math.round(26 * scale);
  ctx.font = `600 ${fontSize}px "Prompt", sans-serif`;
  const textWidth = ctx.measureText(dStr).width;
  const padX = 14 * scale, padY = 8 * scale;
  const boxW = textWidth + padX * 2, boxH = fontSize + padY * 2;
  const posX = center ? (cW - boxW) / 2 : cW - boxW - (16 * scale), posY = cH - boxH - (16 * scale);




  ctx.fillStyle = 'rgba(0, 0, 0, 0.65)';
  if(ctx.roundRect){ ctx.beginPath(); ctx.roundRect(posX, posY, boxW, boxH, 8 * scale); ctx.fill(); }
  else { ctx.fillRect(posX, posY, boxW, boxH); }




  ctx.fillStyle = '#ffffff';
  ctx.textBaseline = 'middle';
  ctx.fillText(dStr, posX + padX, posY + (boxH / 2));
  ctx.restore();
}

function drawCover(ctx, img, x, y, w, h){
  const a = img.width / img.height, ta = w / h;
  let sx = 0, sy = 0, sw = img.width, sh = img.height;
  if(a > ta){ sw = img.height * ta; sx = (img.width - sw) / 2; }
  else { sh = img.width / ta; sy = (img.height - sh) / 2; }
  ctx.drawImage(img, sx, sy, sw, sh, x, y, w, h);
}
function drawContain(ctx, img, x, y, w, h){
  const a = img.width / img.height, ta = w / h;
  let dw = w, dh = h, dx = x, dy = y;
  if(a > ta){ dh = w / a; dy = y + (h - dh) / 2; }
  else { dw = h * a; dx = x + (w - dw) / 2; }
  ctx.drawImage(img, dx, dy, dw, dh);
}
const FILTER_OK = (() => {
  try {
    const c = document.createElement('canvas'); c.width = c.height = 1;
    const x = c.getContext('2d'); if(!('filter' in x)) return false;
    x.filter = 'invert(1)'; x.fillRect(0, 0, 1, 1);
    return x.getImageData(0, 0, 1, 1).data[0] === 255;
  } catch(e){ return false; }
})();
function pixelFilter(ctx, x, y, w, h, css){
  x = Math.round(x); y = Math.round(y); w = Math.round(w); h = Math.round(h);
  const im = ctx.getImageData(x, y, w, h), d = im.data;
  const fx = [...css.matchAll(/(\w+)\((\d+)%\)/g)].map(m => [m[1], m[2] / 100]);
  for(let i = 0; i < d.length; i += 4){
    let r = d[i], g = d[i+1], b = d[i+2];
    for(const [f, v] of fx){
      const l = .299*r + .587*g + .114*b;
      if(f === 'brightness'){ r *= v; g *= v; b *= v; }
      else if(f === 'contrast'){ r = (r-128)*v+128; g = (g-128)*v+128; b = (b-128)*v+128; }
      else if(f === 'saturate'){ r = l+(r-l)*v; g = l+(g-l)*v; b = l+(b-l)*v; }
      else if(f === 'grayscale'){ r += (l-r)*v; g += (l-g)*v; b += (l-b)*v; }
      else if(f === 'sepia'){
        const R = .393*r+.769*g+.189*b, G = .349*r+.686*g+.168*b, B = .272*r+.534*g+.131*b;
        r += (R-r)*v; g += (G-g)*v; b += (B-b)*v;
      }
    }
    d[i] = r; d[i+1] = g; d[i+2] = b;
  }
  ctx.putImageData(im, x, y);
}
function drawFiltered(ctx, img, x, y, w, h, fit, filt){
  if(filt === 'none' || FILTER_OK){
    ctx.save(); if(filt !== 'none') ctx.filter = filt;
    drawImageFit(ctx, img, x, y, w, h, fit); ctx.restore();
  } else { drawImageFit(ctx, img, x, y, w, h, fit); pixelFilter(ctx, x, y, w, h, filt); }
}
function drawImageFit(ctx, img, x, y, w, h, mode){
  if(mode === 'contain') drawContain(ctx, img, x, y, w, h);
  else drawCover(ctx, img, x, y, w, h);
}




function drawTextOverlay(ctx, cW, cH, t1, t2, pos){
  if(pos === 'none' || (!t1 && !t2)) return;
  ctx.save();
  ctx.textAlign = 'center';
  ctx.lineWidth = 14;
  ctx.lineJoin = 'round';
  ctx.strokeStyle = '#000';
  ctx.fillStyle = '#fff';




  const scale = Math.min(cW, cH) / 1200;
  const outline = (text, x, y) => {
    const m = ctx.measureText(text).width, mx = cW - 60;
    if(m > mx) ctx.font = ctx.font.replace(/(\d+)px/, (_, n) => Math.floor(n * mx / m) + 'px');
    ctx.strokeText(text, x, y); ctx.fillText(text, x, y);
  };




  if(pos === 'center'){
    const f1Size = Math.round(74 * scale), f2Size = Math.round(70 * scale), gap = Math.round(46 * scale);
    ctx.textBaseline = 'middle';
    ctx.font = `bold ${f1Size}px "Prompt", sans-serif`;
    if(t1 && t2){
      outline(t1, cW / 2, cH / 2 - gap);
      ctx.font = `bold ${f2Size}px "Prompt", sans-serif`;
      outline(t2, cW / 2, cH / 2 + gap);
    } else {
      outline(t1 || t2, cW / 2, cH / 2);
    }
  } else if(pos === 'topbottom'){
    ctx.font = `bold ${Math.round(64 * scale)}px "Prompt", sans-serif`;
    if(t1){ ctx.textBaseline = 'top'; outline(t1, cW / 2, 35 * scale); }
    if(t2){ ctx.textBaseline = 'bottom'; outline(t2, cW / 2, cH - 35 * scale); }
  }
  ctx.restore();
}

function getGridSize(n, ratio){
  if(n === 1) return [1, 1];
  if(n === 2) return ratio === '3:4' ? [1, 2] : [2, 1];
  if(n <= 4) return [2, 2];
  if(n <= 6) return [3, 2];
  if(n <= 9) return [3, 3];
  const cols = Math.ceil(Math.sqrt(n));
  return [cols, Math.ceil(n / cols)];
}

function getExportCanvas(cid){
  const c = $(cid);
  if(storageGet('out_size', 'normal') !== 'small') return c;
  const k = 800 / Math.max(c.width, c.height);
  if(k >= 1) return c;
  const t = document.createElement('canvas');
  t.width = Math.round(c.width * k); t.height = Math.round(c.height * k);
  t.getContext('2d').drawImage(c, 0, 0, t.width, t.height);
  return t;
}
const shareBlobs = {};
function cacheBlob(cid){ shareBlobs[cid] = null; getExportCanvas(cid).toBlob(b => { shareBlobs[cid] = b; }, 'image/jpeg', 0.9); }
const safeName = (iid, def) => ($(iid)?.value || def).replace(/[^a-zA-Z0-9ก-๙]/g, '_');


async function shareImage(cid, iid, def){
  const v = safeName(iid, def), blob = shareBlobs[cid];
  if(!blob) return downloadImage(cid, iid, def);
  const file = new File([blob], `${def}_${v}.jpg`, { type: 'image/jpeg' });
  if(navigator.canShare && navigator.canShare({ files: [file] })){
    try {
      await navigator.share({ files: [file], title: 'รายงานตู้สินค้า', text: `ส่งภาพ: ${v}` });
      recordDownload(); maybeBackup(getExportCanvas(cid), def, v);
      showToast('📲 แชร์รูปสำเร็จเรียบร้อย');
    } catch(e){ if(e.name !== 'AbortError') downloadImage(cid, iid, def); }
  } else downloadImage(cid, iid, def);
}


async function backupToDrive(canvas, folderName, fileName){
  try {
    const res = await fetch(GAS_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ folderName, fileName, image: canvas.toDataURL('image/jpeg', 0.85) })
    });
    const data = await res.json();
    if(data && data.status === 'success') showToast('☁️ สำรองรูปลง Google Drive แล้ว');
  } catch(e){}
}


function maybeBackup(c, def, v){
  if(!isBackupOn()) return;
  const targetFolder = (def === 'รายงาน_ตู้') ? 'Vending_เติมตู้' : 'Vending_รูปทั่วไป';
  const n = new Date();
  backupToDrive(c, targetFolder, `${n.getFullYear()}-${pad2(n.getMonth()+1)}-${pad2(n.getDate())}_${pad2(n.getHours())}-${pad2(n.getMinutes())}_${v}_${Math.random().toString(36).substring(2,6)}.jpg`);
}


function downloadImage(cid, iid, def){
  const c = getExportCanvas(cid), v = safeName(iid, def);
  c.toBlob(blob => {
    if(!blob) return showToast('⚠️ สร้างไฟล์ไม่สำเร็จ ลองใหม่อีกครั้ง');
    const u = URL.createObjectURL(blob), a = document.createElement('a');
    a.download = `${def}_${v}.jpg`; a.href = u;
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(u), 5000);
    recordDownload(); maybeBackup(c, def, v);
  }, 'image/jpeg', 0.90);
}

let pdfImgs = [], pdfLoading = false;
async function handlePdfFiles(e){
  const files = Array.from(e.target.files);
  if(!files.length) return;
  pdfLoading = true;
  $('pdfCountText').innerText = `กำลังโหลด ${files.length} รูป...`;
  try {
    const loaded = await Promise.all(files.map(loadImageFile));
    pdfImgs.forEach(releaseImage);
    pdfImgs = loaded;
    $('pdfCountText').innerText = `เลือกแล้ว ${loaded.length} รูป`;
  } catch(err){ alert(err.message); }
  finally { pdfLoading = false; }
}




function generatePdf(){
  if(pdfLoading || !pdfImgs.length) return alert('กรุณาเลือกรูปภาพก่อนครับ');
  recordDownload();
  const doc = new jspdf.jsPDF({ orientation: $('pdfOrientation').value, unit: 'mm', format: 'a4' });
  const pw = doc.internal.pageSize.getWidth(), ph = doc.internal.pageSize.getHeight();
  const mw = pw - 20, mh = ph - 20;
  pdfImgs.forEach((img, idx) => {
    if(idx > 0) doc.addPage();
    const ratio = img.width / img.height;
    let w = mw, h = mw / ratio;
    if(h > mh){ h = mh; w = mh * ratio; }
    const c = document.createElement('canvas');
    c.width = img.width; c.height = img.height;
    c.getContext('2d').drawImage(img, 0, 0);
    doc.addImage(c.toDataURL('image/jpeg', 0.90), 'JPEG', (pw - w) / 2, (ph - h) / 2, w, h);
  });
  doc.save('เอกสาร.pdf');
}

function renderMonthlyStats(){
  const sched = getStoredSchedule(), prefix = `${currentCalYear}-${pad2(currentCalMonth+1)}`;
  let t8 = 0, t9 = 0, p8 = {}, p9 = {}, set = new Set();
  Object.keys(sched).forEach(k => {
    if(k.startsWith(prefix)){
      (sched[k].v8 || []).filter(Boolean).forEach(n => { t8++; p8[n] = (p8[n]||0)+1; set.add(n); });
      (sched[k].v9 || []).filter(Boolean).forEach(n => { t9++; p9[n] = (p9[n]||0)+1; set.add(n); });
    }
  });
  $('statTotalRounds').innerText = `รวม ${t8+t9} รอบ (V8: ${t8} | V9: ${t9})`;
  const inc8 = parseFloat($('inpIncomeV8').value) || 0, inc9 = parseFloat($('inpIncomeV9').value) || 0;
  const wrap = $('peopleSummaryTableWrap');
  if(!set.size){ lastSummaryText = ''; wrap.innerHTML = '<p style="font-size:0.8rem;color:var(--mut);text-align:center;margin-top:6px">ยังไม่มีข้อมูล</p>'; return; }
  const list = Array.from(set).map(n => {
    const c8 = p8[n] || 0, c9 = p9[n] || 0;
    const sum = (t8 > 0 ? (c8/t8)*inc8 : 0) + (t9 > 0 ? (c9/t9)*inc9 : 0);
    return { n, c8, c9, tot: c8+c9, sum };
  }).sort((a,b) => b.sum - a.sum || b.tot - a.tot);




  lastSummaryText = `📅 สรุปรอบเติมตู้ ${THAI_MONTHS[currentCalMonth]} ${currentCalYear + 543}\nรวม ${t8+t9} รอบ (V8: ${t8} | V9: ${t9})\n` +
    list.map(x => `• ${x.n}: V8 ${x.c8} | V9 ${x.c9} | รวม ${x.tot} รอบ` + ((inc8 || inc9) ? ` = ${Math.round(x.sum).toLocaleString()} บาท` : '')).join('\n');
  wrap.innerHTML = `<table><thead><tr><th>ผู้เติม</th><th>V8</th><th>V9</th><th>รวม</th><th>ยอดเงิน</th></tr></thead><tbody>` +
    list.map(x => `<tr><td><strong>${esc(x.n)}</strong></td><td style="color:#f97316">${x.c8}</td><td style="color:#22c55e">${x.c9}</td><td>${x.tot}</td><td style="color:var(--pri);font-weight:700">${(inc8||inc9)?x.sum.toLocaleString(undefined,{maximumFractionDigits:0})+' บ.':'-'}</td></tr>`).join('') + `</tbody></table>`;
}

  list.forEach((n, i) => {
    chip(n, () => { $('txt2_branch').value = n; $('txt2_branch').dispatchEvent(new Event('input')); });
    chip('✕', () => { list.splice(i, 1); storageSet(BR_KEY, JSON.stringify(list)); renderBranchChips(); });
  });
  chip('＋ บันทึกชื่อนี้', () => {
    const v = $('txt2_branch').value.trim();
    if(!v || list.includes(v)) return showToast('ไม่มีชื่อใหม่ให้บันทึก');
    list.push(v); storageSet(BR_KEY, JSON.stringify(list)); renderBranchChips();
  });
}

async function copySummary(){
  if(!lastSummaryText) return showToast('ยังไม่มีข้อมูลให้คัดลอก');
  try { await navigator.clipboard.writeText(lastSummaryText); }
  catch(e){ const t = document.createElement('textarea'); t.value = lastSummaryText; document.body.appendChild(t); t.select(); document.execCommand('copy'); t.remove(); }
  showToast('📋 คัดลอกแล้ว วางในไลน์ได้เลย');
}
